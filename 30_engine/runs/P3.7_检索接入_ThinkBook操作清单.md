## 2. 给 `tools/kb_import.py` 打补丁

**补丁 A：在 `def import_kb(` 之前插入新函数**
```python
def build_jsonl_pwkb(kb_name: str, src_dir: Path,
                     meta: dict | None = None) -> tuple[list[dict], dict]:
    """Public_Writing_KB 公文 QA（jsonl）→ 本库格式（ADR-020 修订：公文库由 PWKB 维护，本项目只读索引）。

    读 src_dir/c[1-6]_*.jsonl，保留原 id（QA-C3-00001，便于回溯与 F6 用例核对）；
    公文专属维度编进 tags（gw:C3 / gw:<小类> / gw:W2 / gw:depth=method），用 search.py --tags 过滤。
    输出是派生文件，每次导入整体重写，禁止手改。
    """
    meta = meta or {}
    files = sorted(src_dir.glob("c[1-6]_*.jsonl"))
    out: list[dict] = []
    per_file: dict[str, int] = {}
    for f in files:
        n = 0
        for line in f.read_text(encoding="utf-8").splitlines():
            if not line.strip():
                continue
            r = json.loads(line)
            if r.get("status", "active") != "active":
                continue
            gw = [f"gw:C{r['category_l1']}", f"gw:{r['category_l2']}"]
            gw += [f"gw:{w}" for w in r.get("w_layer") or []]
            if r.get("depth"):
                gw.append(f"gw:depth={r['depth']}")
            out.append({
                "id": r["id"],
                "kb": kb_name,
                "layer": meta.get("layer", ""),
                "category": meta.get("category", ""),
                "class": meta.get("class", ""),
                "sensitive": bool(meta.get("sensitive")),
                "questions": [q.strip() for q in r["question"].split(" / ") if q.strip()],
                "answer": r["answer"],
                "tags": ", ".join(list(r.get("tags") or []) + gw),
                "scenarios": r.get("scenarios", ""),
                "keywords": ", ".join(r.get("question_keywords") or []),
                "answer_hash": r.get("answer_hash") or answer_hash(r["answer"]),
                "source": {"file": f.name, "anchor": r.get("source_anchor", ""), "pwkb_id": r["id"]},
                "status": "active",
                "imported_at": date.today().isoformat(),
            })
            n += 1
        per_file[f.name] = n
    stats = {"files": len(files), "records": len(out), "duplicates_skipped": 0, "warnings": [],
             "per_file": per_file, "question_variants": sum(len(r["questions"]) for r in out)}
    return out, stats
```

**补丁 B：替换 `import_kb` 开头到"records, stats = …"这一段为**（关键：分流必须在"找不到 xlsx 就退出"之前）
```python
def import_kb(registry: dict, kb_name: str, dry_run: bool) -> dict:
    kbs = registry.get("kbs") or {}
    if kb_name not in kbs:
        raise SystemExit(f"失败：知识库未注册：{kb_name}")
    cfg = kbs[kb_name] or {}
    target_dir = KBS_ROOT / kb_name
    assert_target_safe(target_dir)          # 红线一
    info = kb_registry.kb_layer_info(registry, kb_name)
    meta = {**info, "sensitive": bool(cfg.get("sensitive"))}

    if cfg.get("format") == "pwkb_jsonl":   # 公文库：只读 Public_Writing_KB，不走 xlsx
        src_dir = Path(str(cfg.get("source")))
        if not src_dir.is_dir():
            raise SystemExit(f"失败：{kb_name} 源目录不存在：{src_dir}（设 KBRAG_PWKB_ROOT？）")
        records, stats = build_jsonl_pwkb(kb_name, src_dir, meta)
    else:
        files = kb_registry.kb_sources(cfg)
        if not files:
            raise SystemExit(f"失败：{kb_name} 没有找到任何 xlsx 源文件（source={cfg.get('source')}）")
        records, stats = build_jsonl(kb_name, files, meta)
```
